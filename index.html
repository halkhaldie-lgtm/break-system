function showCustomAlert(message) { // التحقق مما إذا كان هناك تنبيه قديم وإزالته const existingAlert = document.getElementById('custom-alert-box'); if (existingAlert) { existingAlert.remove(); } // إنشاء عنصر الحاوية للتنبيه const alertBox = document.createElement('div'); alertBox.id = 'custom-alert-box'; alertBox.style.cssText = ` position: fixed; top: 20px; left: 50%; transform: translateX(-50%); background-color: #fff; border: 2px solid #ff4d4d; box-shadow: 0 4px 15px rgba(0, 0, 0, 0.2); padding: 20px 25px; z-index: 10000; border-radius: 8px; text-align: center; max-width: 400px; width: 90%; font-family: Arial, sans-serif; `; // محتوى التنبيه alertBox.innerHTML = `
⚠️ تنبيه
${message}
`; document.body.appendChild(alertBox); // حدث زر الإغلاق document.getElementById('custom-alert-btn').onclick = function() { alertBox.remove(); }; }
