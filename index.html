function endBreak(index) { initAudioContext(); const finishedItem = state.active.splice(index, 1)[0]; if (finishedItem) { const endTime = Date.now(); const durationMs = endTime - finishedItem.timestamp; const durationMins = Math.floor(durationMs / 60000); const durationText = durationMins === 0 ? 'Less than a minute' : `${durationMins} mins`; const endTimeStr = new Date().toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }); const breakDateStr = finishedItem.dateStr || new Date().toLocaleDateString('en-GB', { day: '2-digit', month: '2-digit', year: 'numeric' }); let typeText = 'Normal Break'; if (finishedItem.type === 'quick') typeText = 'Quick Break (5m)'; else if (finishedItem.type === 'request') typeText = 'Requested Break'; let isOvertime = false; if (finishedItem.type === 'quick' && durationMins > 5) { isOvertime = true; } else if ((finishedItem.type === 'normal' || finishedItem.type === 'request') && durationMins > 20) { isOvertime = true; } state.reports.unshift({ name: finishedItem.name, type: typeText, dateStr: breakDateStr, startTime: finishedItem.timeStr, endTime: endTimeStr, duration: durationText, isOvertime: isOvertime, endTimestamp: endTime }); // تعديل منطق الإرجاع للطابور بناءً على طلبك if (finishedItem.keepPositionOnReturn) { delete finishedItem.keepPositionOnReturn; finishedItem.timestamp = Date.now(); finishedItem.dateStr = new Date().toLocaleDateString('en-GB', { day: '2-digit', month: '2-digit', year: 'numeric' }); // إذا كان لديه رقم مخزن مسبقاً (مثلاً رقم 3)، نحاول إرجاعه لنفس رقمه أو حسب الطابور الحالي let targetIndex = finishedItem.originalQueueIndex !== undefined ? finishedItem.originalQueueIndex : 0; // التأكد من أن الترتيب لا يتجاوز حجم القائمة الحالية if (targetIndex > state.waiting.length) { targetIndex = state.waiting.length; } state.waiting.splice(targetIndex, 0, finishedItem); delete finishedItem.originalQueueIndex; } } db.set(state); }
